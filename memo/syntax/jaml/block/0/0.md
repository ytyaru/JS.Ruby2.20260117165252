# テキストブロックパーサを作りたい

JavaScriptにおいて原稿テキストからHTMLテキストに変換するパーサを作りたい。このうち大枠であるブロック要素の抽象的な部分だけを作りたい。

```js
class Jaml {
    parse(text) {
        const htmls = [];
        const blocks = Block.lex(Manuscript.normalize(text));
        for (let block of blocks) {
            const inlines = Inline.lex(block.inlines);
            htmls.push(Block.toHtml(block, Inline.toHtml(inlines)));
        }
        return htmls.join('\n');
    }
}
```
```js
class Manuscript {
    static normalize(text) {
        return text.normalize('NFC').replaceAll(/\r|\r\n/, '\n').replaceAll('\0', '');
    }
}
```
