# 不可視文字の取得

Unicodeには不可視文字が大量にある。邪魔になる場合があるため、不可視文字を特定したい。残念ながらフォントグリフとの兼ね合いにより画一的な方法がない。そこで以下のような方法で不可視文字を抽出する。

1. Unicode公式データからの抽出 (unicode.tsv)
2. 人力抽出リスト（heuristic.tsv）の作成方法

## 作成日時とUnicodeバージョン

* 2026-01-30
* Unicode version 17.0

文字セットのコードポイントやカテゴリ、プロパティの名前などはUnicodeのバージョンによって変わるため注意。未来の保証は誰にもできない。

## 1. Unicode公式データからの抽出 (unicode.tsv)

Unicode標準のデータベース（UCD: Unicode Character Database）から、一般的に「不可視」とみなされるカテゴリの文字を抽出します。

### 判定基準（採用カテゴリ）

カテゴリとプロパティに関する以下の条件を一つでも満たしたものを「Unicode定義による不可視文字」とする

#### カテゴリ

`General_Category`プロパティが以下の値を持つ文字を「不可視文字」の候補として抽出します。

* `Cc` (Control): 制御文字（改行、タブなど）
* `Cf` (Format): 書式設定文字（ZWJ、BOM、RLOなど）
* `Cs` (Surrogate): サロゲートペア文字（ペアで一文字を成すための特殊なコード。単独では表示されない）
* `Zs` (Space Separator): 空白文字（スペース、全角スペースなど）
* `Zl` (Line Separator): 行区切り
* `Zp` (Paragraph Separator): 段落区切り
* `Mn` (Nonspacing Mark): 非前進結合記号（基底文字に重なる不可視の修飾など）

#### プロパティ

* `Default_Ignorable_Code_Point`プロパティが`true`である
* `White_Space`プロパティが`true`である

### 抽出用Pythonコード (extract_invisible_chars_from_unicode_to_tsv.py)

> 以下のコードを実行することで、最新の UnicodeData.txt をオンラインから取得し、指定カテゴリの文字を unicode.tsv に出力します。

```python
import urllib.request
import os
import re
from collections import defaultdict

def download_ucd_file(filename, version="latest"):
    """
    Unicode Character Database (UCD)からファイルをダウンロードする。
    """
    base_url = f"https://www.unicode.org/Public/UCD/{version}/ucd/"
    url = base_url + filename
    if not os.path.exists(filename):
        print(f"{filename} をダウンロード中...")
        try:
            urllib.request.urlretrieve(url, filename)
            print("ダウンロード完了。")
        except urllib.error.URLError as e:
            print(f"エラー: {filename} のダウンロードに失敗しました。")
            print(e)
            return False
    return True

def parse_ucd_range_file(filename, property_name):
    """
    コードポイントの範囲とプロパティが記述されたUCDファイルをパースする。
    例: PropList.txt, DerivedCoreProperties.txt
    """
    codepoints = set()
    if not download_ucd_file(filename):
        return codepoints

    with open(filename, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith('#'):
                continue
            
            # コメント部分を除去
            if '#' in line:
                line = line.split('#')[0].strip()

            parts = [p.strip() for p in line.split(';')]
            if len(parts) < 2 or parts[1] != property_name:
                continue

            codepoint_range = parts[0]
            if '..' in codepoint_range:
                start, end = [int(p, 16) for p in codepoint_range.split('..')]
                for cp in range(start, end + 1):
                    codepoints.add(cp)
            else:
                codepoints.add(int(codepoint_range, 16))
    
    print(f"{filename} から {property_name} プロパティを持つ文字を {len(codepoints)} 個読み込みました。")
    return codepoints

def get_general_categories():
    """
    UnicodeData.txt をパースして、各コードポイントのGeneral_Categoryを取得する。
    """
    filename = "UnicodeData.txt"
    categories = {}
    if not download_ucd_file(filename):
        return categories

    with open(filename, 'r', encoding='utf-8') as f:
        for line in f:
            parts = line.strip().split(';')
            if len(parts) > 2:
                codepoint = int(parts[0], 16)
                name = parts[1]
                category = parts[2]
                
                # 範囲指定の開始文字 (<..., First>)
                if name.endswith(", First>"):
                    # 次の行が範囲の終了文字のはず
                    next_line = next(f, None)
                    if next_line:
                        end_parts = next_line.strip().split(';')
                        end_codepoint = int(end_parts[0], 16)
                        for cp in range(codepoint, end_codepoint + 1):
                            categories[cp] = category
                else:
                    categories[codepoint] = category
    
    print(f"{filename} からGeneral_Category情報を読み込みました。")
    return categories

def main():
    """
    メイン処理
    """
    # --- 抽出条件の定義 ---
    target_categories = {
        'Cc',  # Other, Control
        'Cf',  # Other, Format
        'Cs',  # Other, Surrogate
        'Cn',  # Other, Not Assigned
        'Mn',  # Mark, Nonspacing
        'Zl',  # Separator, Line
        'Zp',  # Separator, Paragraph
        'Zs',  # Separator, Space
    }

    # --- データの準備 ---
    # 各プロパティを持つコードポイントのセットを取得
    white_space_cps = parse_ucd_range_file("PropList.txt", "White_Space")
    # DerivedCoreProperties.txt にも White_Space の定義がある場合がある
    white_space_cps.update(parse_ucd_range_file("DerivedCoreProperties.txt", "White_Space"))
    
    def_ignorable_cps = parse_ucd_range_file("DerivedCoreProperties.txt", "Default_Ignorable_Code_Point")
    
    # 全コードポイントのGeneral_Categoryを取得
    general_categories = get_general_categories()

    # --- 不可視文字の抽出 ---
    invisible_chars = set()
    
    # Unicodeの全範囲 (0x000000 - 0x10FFFF) をチェック
    for cp in range(0x110000):
        # 条件1: General_Category が対象のいずれか
        category = general_categories.get(cp, 'Cn') # 未定義はCnとして扱う
        if category in target_categories:
            invisible_chars.add(cp)
            continue
            
        # 条件2: White_Space プロパティが true
        if cp in white_space_cps:
            invisible_chars.add(cp)
            continue

        # 条件3: Default_Ignorable_Code_Point プロパティが true
        if cp in def_ignorable_cps:
            invisible_chars.add(cp)
            continue

    # --- ファイルへの出力 ---
    output_filename = "invisible-chars.txt"
    sorted_invisible_chars = sorted(list(invisible_chars))
    
    try:
        with open(output_filename, 'w', encoding='utf-8', newline='\n') as f:
            for cp in sorted_invisible_chars:
                # コードポイントを16進数文字列（大文字、プレフィックスなし）に変換
                hex_cp = format(cp, 'X')
                f.write(f"{hex_cp}\n")
        
        print(f"\n抽出した不可視文字 {len(sorted_invisible_chars)} 個を {output_filename} に出力しました。")

    except IOError as e:
        print(f"エラー: ファイル '{output_filename}' の書き込みに失敗しました。")
        print(e)

if __name__ == "__main__":
    main()
```

### 取得できない不可視文字について

Unicodeのカテゴリやプロパティで多くの不可視文字を取得できるが、全部ではない。

ここで取得できない不可視文字は以下のようなものがある。

* `Co`(私用): いわゆる外字。
* `Cn`(未定義): Unicodeにおける未定義。
* 他: カテゴリでは判断できない。可視文字が含まれたカテゴリ内に点在する不可視文字がある。これらは[Unicode charts][]から視認することで人力抽出するしかない

#### `Co`(私用)

ユーザが独自に外字を用意できる領域。自作フォントを用意すれば可視文字を用意できる。さもなくば不可視文字である。あるいはユーザが意識しておらず、何らかのアプリが自動的に用意している場合もある。

いずれにせよ、以下のような設定をすることで、可視／不可視文字として分類する必要がある。そのためのAPIを考案し実装する必要があるだろう。

* 使用する: 可視として扱う
* 使用しない: 不可視として扱う
* 未定義: 可視／不可視どちらとして扱うか

```javascript
class CoChars {
    constructor(used, unused, isOtherVisibled=false) {
        this._ = {
            used: new CharGroup(used, {ja:'使用する私用領域', en:'used Co'}), 
            unused: new CharGroup(unused, {ja:'使用しない私用領域', en:'unused Co'})
            other: new CharGroup(/*上記以外の私用領域全部*/, {ja:'使用しない私用領域', en:'unused Co'}};
    }
    get visibles() {}// 可視文字を返す
    get invisibles() {}// 不可視文字を返す
}
```

これついては**未実装**であり今後の課題である。

#### `Cn`(未定義):

これについてはどう扱うべきか不明。不可視文字ではあろう。しかしUnicodeのバージョン次第では可視の可能性もある。おそらく未定義は以下のような理由だと思われる。

* 将来の拡張のためにスペースを空けてある
* キリの良い番号にするためにスペースを空けてある
* 文字の概念における認識や配置が間違っており欠番になった

いずれにせよ、Unicodeバージョンによって変更しうる可能性がある。これを意識し、実際のシステムと合わせることは極めて困難。よって管理不能と言わざるを得ない。

#### 他:

Unicodeのカテゴリやプロパティだけでは可視性を判断できない文字がある。可視文字が含まれたカテゴリ内に点在する不可視文字がある場合もあるからだ。これらは[Unicode charts][]から視認することで人力抽出するしかない。膨大な字数があるため人間業ではない。また、Unicodeのバージョンが変わるたびに変更される恐れがあるためキリがない。

よってこの領域に関しては、以降の「人力抽出リスト」として管理することにする。

## 2. 人力抽出リスト（heuristic.tsv）の作成方法

自動抽出では網羅できない、あるいは「実態として不可視」として扱われる文字を、外部の情報源と目視確認（PDFチャート）を組み合わせて定義します。

### 2.1. ブログ記事からの初期抽出

まず、以下の2つの技術記事から、不可視文字として紹介されているコードポイントを抽出します。

- `1.tsv`: [Unicode スペース以外の見えない空白文字の一覧][]
- `2.tsv`: [gigazine.net: invisible-characters][]

これらをマージした中間リスト（`3.tsv`相当）をPythonスクリプトで作成する。

### 2.2. Unicode公式チャートによる目視検証

上記で抽出した文字について、[Unicode公式のCode Charts](https://www.unicode.org)（PDF）を確認し、実態が「不可視」であるかを一文字ずつ判定します。

- **判断の根拠**:
  - **不可視判定**: PDF内で「点線の四角枠」で囲まれており、図形（グリフ）が描かれていないもの。これらはそのままリストに維持、または新規追加する。
  - **可視判定**: PDF内で「実線の黒い図形」が描かれているもの。これらは専用フォントがあれば表示されるべき文字であるため、リストから除外する。

検証に使用したPDFは以下の通りです。

* [U1100.pdf][]
* [U1780.pdf][]
* [U2000.pdf][]
* [U2800.pdf][]
* [U3000.pdf][]
* [U3130.pdf][]
* [UFE70.pdf][]
* [UFF00.pdf][]
* [U1D100.pdf][]

理想的にはUnicode定義された全字をPDFで目視検査すべきだが、数が多すぎるためブログ記事から抽出された字に関連する範囲のみを対象とする。

#### 検証結果の反映

以下ファイルを手動で作成する。

- **除外 (`3-1.tsv`)**:
  - `0x2800` (BRAILLE BLANK): PDF上で微小な8つの点（図形）が確認されたため、本来は可視文字として扱う。
- **追加 (`3+1.tsv`)**:
  - ブログ記事に漏れていたが、PDFで点線枠であることを確認した以下の文字を補完。
    - `0x2011`, `0x2028`〜`0x202E`, `0x2066`〜`0x2069`, `0x180B`〜`0x180D`, `0x180F`

### 2.3. 最終的な heuristic.tsv の生成

ブログ抽出分から「可視判定分`3-1.tsv`」を削除し、「漏れ補完分`3+1.tsv`」を追加したものを `4.tsv` とし、これを **`src/db/heuristic.tsv`** として確定させます。

### 2.4. 抽出・加工用Pythonコード例

#### `1〜3.tsv`作成

```python
#!/usr/bin/env python3
import requests
import re

def extract_unicode_from_url(url):
    print(f"取得中: {url}")
    try:
        # ブラウザからのアクセスを装う
        headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'}
        response = requests.get(url, headers=headers, timeout=15)
        response.encoding = response.apparent_encoding
        
        # HTMLソース全体から U+XXXX 形式を抽出
        found_codes = re.findall(r'[uU]\+([0-9a-fA-F]{4,6})', response.text)
        
        unique_codes = {int(c, 16) for c in found_codes}
        print(f"発見数: {len(unique_codes)} 文字 ({url})")
        return unique_codes
    except Exception as e:
        print(f"エラー発生 ({url}): {e}")
        return set()

def save_tsv(filename, code_set):
    # 16進数値のみを昇順で保存
    with open(filename, 'w', encoding='utf-8') as f:
        for code in sorted(list(code_set)):
            f.write(f"0x{code:04X}\n")

def format_js_array(code_set):
    sorted_codes = sorted(list(code_set))
    if not sorted_codes: return "[]"
    
    ranges = []
    start = sorted_codes[0]
    end = sorted_codes[0]
    
    for i in range(1, len(sorted_codes)):
        if sorted_codes[i] == end + 1:
            end = sorted_codes[i]
        else:
            ranges.append(f"0x{start:04X}" if start == end else f"[0x{start:04X}, 0x{end:04X}]")
            start = end = sorted_codes[i]
    ranges.append(f"0x{start:04X}" if start == end else f"[0x{start:04X}, 0x{end:04X}]")
    
    return "[" + ", ".join(ranges) + "]"

if __name__ == "__main__":
    # 対象URL
    url1 = "https://tech-blog.s-yoshiki.com/entry/303"
    url2 = "https://gigazine.net/gsc_news/en/20221208-invisible-characters/"
    
    set1 = extract_unicode_from_url(url1)
    set2 = extract_unicode_from_url(url2)
    
    # 1.tsv, 2.tsv の作成
    save_tsv("1.tsv", set1)
    save_tsv("2.tsv", set2)
    
    # 重複排除マージと 3.tsv の作成
    merged = set1.union(set2)
    save_tsv("3.tsv", merged)
    
    # JSファイルの生成
    js_array_str = format_js_array(merged)
    js_content = (
        f'import {{CharGroup}} from "./char-group.js"\n'
        f'export const heuristicInvisibleChars = new CharGroup({js_array_str}, '
        f'{{ja:\'人力抽出した不可視文字\', en:\'HeuristicInvisibleChars\'}});\n'
    )
    
    with open("heuristic-invisible-chars.js", "w", encoding="utf-8") as f:
        f.write(js_content)
    
    print(f"\n完了: 1.tsv, 2.tsv, 3.tsv, および heuristic-invisible-chars.js を作成しました。")
    print(f"合計ユニーク文字数: {len(merged)}")
```

#### `4.tsv`作成

```python
#!/usr/bin/env python3

def main():
    # 1. 既存の3.tsvを読み込む
    try:
        with open("3.tsv", "r", encoding="utf-8") as f:
            # 0xXXXX 形式の行から数値セットを作成
            current_codes = {int(line.strip(), 16) for line in f if line.strip().startswith("0x")}
    except FileNotFoundError:
        print("エラー: 3.tsv が見つかりません。")
        return

    # 2. 削除要素 (3-1.txt相当): PDFで実線図形があったもの
    to_remove = {0x2800}
    
    # 3. 追加要素 (3+1.txt相当): PDFで点線枠を確認した漏れ分
    to_add = {
        0x2011, 0x2028, 0x2029, 0x202A, 0x202B, 0x202C, 0x202D, 0x202E,
        0x2066, 0x2067, 0x2068, 0x2069,
        0x180B, 0x180C, 0x180D, 0x180F
    }

    # 4. 集合演算で更新
    final_set = (current_codes - to_remove) | to_add

    # 5. 4.tsv (最終結果) の出力
    with open("4.tsv", "w", encoding="utf-8") as f:
        for code in sorted(list(final_set)):
            f.write(f"0x{code:04X}\n")

    # 6. JSソースの生成 (heuristic-invisible-chars.js)
    js_array_str = format_js_array(final_set)
    js_content = (
        f'import {{CharGroup}} from "./char-group.js"\n'
        f'export const heuristicInvisibleChars = new CharGroup({js_array_str}, '
        f'{{ja:\'人力抽出した不可視文字\', en:\'HeuristicInvisibleChars\'}});\n'
    )
    
    with open("heuristic-invisible-chars.js", "w", encoding="utf-8") as f:
        f.write(js_content)

    print(f"完了: 4.tsv を作成しました (合計 {len(final_set)} 文字)")
    print(f"除外: {[hex(c) for c in to_remove]}")
    print(f"追加: {len(to_add)} 文字を補完しました")

def format_js_array(code_set):
    sorted_codes = sorted(list(code_set))
    if not sorted_codes: return "[]"
    ranges, start, end = [], sorted_codes[0], sorted_codes[0]
    for i in range(1, len(sorted_codes)):
        if sorted_codes[i] == end + 1:
            end = sorted_codes[i]
        else:
            ranges.append(f"0x{start:04X}" if start == end else f"[0x{start:04X}, 0x{end:04X}]")
            start = end = sorted_codes[i]
    ranges.append(f"0x{start:04X}" if start == end else f"[0x{start:04X}, 0x{end:04X}]")
    return "[" + ", ".join(ranges) + "]"

if __name__ == "__main__":
    main()
```
